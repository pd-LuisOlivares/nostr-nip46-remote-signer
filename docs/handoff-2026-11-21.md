```powershell
# Instalación (PowerShell 7, raíz del repositorio)
py -3.12 -m venv .venv; .\.venv\Scripts\Activate.ps1
python -m pip install -c constraints.txt -e ".[dev]"

# Aplicación local. PRIMER ARRANQUE: la contraseña inicial del administrador se pasa por entrada oculta (nunca en la línea de comandos):
$env:ADMIN_PASSWORD = Read-Host "Contraseña inicial" -MaskInput      # 12 a 128 caracteres; sin valor por defecto

Remove-Item Env:ADMIN_PASSWORD                                   # ya no hace falta: la cuenta persiste en data/nip46.db (excluido de Git)

# Docker (publica solo en 127.0.0.1)
docker compose up --build -d            # primer arranque: requiere ADMIN_PASSWORD (entrada oculta); la base persiste en el volumen app_data
docker compose down                     # conserva el volumen (el administrador); `down -v` lo borra (así se recupera una contraseña perdida)

# Pruebas
python -m pytest -q                                          # todo (E2E de navegador incluidas si hay Chrome/Edge)
python -m pytest -q -m "not e2e"                             # sin navegador
$env:E2E_DOCKER = "1"; python -m pytest -q tests/e2e/test_docker_ndk.py tests/e2e/test_docker_persistence.py   # E2E Docker (NDK real y persistencia de SQLite)
python lab/run_product_regression.py ; python lab/run_product_regression.py --api   # regresión NDK en red interna
```
